# WWE SmackDown vs. Raw 2011 - ReXGlue recompiled PC port
#
#   svr2011            the recompiled game (src/, generated/default/)
#   svr2011_launcher   "SvR2011 Launcher.exe": install from the disc image,
#                      settings, play (launcher/)
#
# SDK boilerplate lives in generated/rexglue.cmake.

cmake_minimum_required(VERSION 3.25)
project(svr2011 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

option(SVR2011_P2P_SESSIONS "Build experimental directory-backed sessions (no gameplay networking)" OFF)
option(SVR2011_SESSION_TESTS "Build session lifecycle/guest ABI tests" OFF)
option(SVR2011_BUILD_GAME "Build the game (disable for asset-free developer tests)" ON)

if(SVR2011_P2P_SESSIONS OR SVR2011_SESSION_TESTS)
    # Use an installed JSON dependency where available; otherwise fetch a
    # pinned upstream commit. Ordinary builds incur no new download/dependency.
    find_package(nlohmann_json 3.11.3 CONFIG QUIET)
    if(NOT TARGET nlohmann_json::nlohmann_json)
        include(FetchContent)
        FetchContent_Declare(svr2011_json
            GIT_REPOSITORY https://github.com/nlohmann/json.git
            GIT_TAG 9cca280a4d0ccf0c08f47a99aa71d1b0e52f8d03) # v3.11.3
        FetchContent_MakeAvailable(svr2011_json)
    endif()
    add_library(svr2011_sessions STATIC src/online_sessions.cpp src/online_directory.cpp)
    target_include_directories(svr2011_sessions PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/src)
    target_link_libraries(svr2011_sessions PRIVATE nlohmann_json::nlohmann_json)
    if(SVR2011_SESSION_TESTS)
        enable_testing()
        add_executable(svr2011_session_tests tests/online_sessions_test.cpp)
        target_link_libraries(svr2011_session_tests PRIVATE svr2011_sessions nlohmann_json::nlohmann_json)
        add_test(NAME online_session_groundwork COMMAND svr2011_session_tests)
    endif()
endif()
if(NOT SVR2011_BUILD_GAME)
    if(NOT SVR2011_SESSION_TESTS)
        message(FATAL_ERROR "Enable SVR2011_SESSION_TESTS when disabling the game build")
    endif()
    return()
endif()

include(generated/rexglue.cmake)

set(SVR2011_SOURCES
    src/main.cpp
    src/svr2011_app.cpp
    src/script_input.cpp
    src/keyboard_typing.cpp
    src/platform.cpp
    src/frame_stats.cpp
    src/fps_overlay.cpp
    src/achievements_page.cpp
    src/graphics_page.cpp
    src/touch_controls.cpp
    src/discord_presence.cpp
    src/menu_hooks.cpp
    src/perf_hooks.cpp
    src/saves.cpp
    src/dlc.cpp
    src/caw_logos.cpp
    src/user_movies.cpp
    src/game_files.cpp
)

# Developer build: count calls into every function of the game's D3D library
# (src/d3d_trace.h). Hooks are generated by tools/gen_d3d_trace.py.
option(SVR2011_D3D_TRACE "Build the D3D library call census" OFF)
if(SVR2011_D3D_TRACE)
    list(APPEND SVR2011_SOURCES src/d3d_trace.cpp src/d3d_trace_hooks.cpp)
else()
    # Native renderer (native_renderer=main), drawing through plume. Its hooks
    # overlap the census hooks, so it is left out of census builds.
    list(APPEND SVR2011_SOURCES
        src/native/gpu.cpp
        src/native/backend_vulkan.cpp
        src/native/recording_list.cpp
        src/native/native_renderer.cpp
        src/native/native_hooks.cpp
        src/native/textures.cpp)
    # The texture layout / tiling code of the SDK's Xenos GPU plugin (not
    # exported by the plugin DLL), so native textures match the emulated ones.
    list(APPEND SVR2011_SOURCES
        ${REXSDK_DIR}/src/graphics/pipeline/texture/util.cpp
        ${REXSDK_DIR}/src/graphics/pipeline/texture/info_formats.cpp)
endif()

# plume (github.com/renderbag/plume, MIT): the native renderer's graphics
# layer - Direct3D 12 now, Vulkan for Android / Linux. Pinned at d723793 with
# port/patches/plume-svr2011.patch applied.
set(PLUME_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../recomp/plume" CACHE PATH "plume source tree")
if(NOT SVR2011_D3D_TRACE)
    add_subdirectory(${PLUME_DIR} plume EXCLUDE_FROM_ALL)
endif()

# Windows only: XAudio2 audio, crash reports, PC music (XAudio2 + Media
# Foundation), the D3D12 renderer backend. Elsewhere posix_stubs.cpp.
if(WIN32)
    list(APPEND SVR2011_SOURCES src/xaudio2_audio.cpp src/crash_report.cpp src/music.cpp)
    if(NOT SVR2011_D3D_TRACE)
        list(APPEND SVR2011_SOURCES src/native/backend_d3d12.cpp)
    endif()
else()
    list(APPEND SVR2011_SOURCES src/posix_stubs.cpp)
endif()

if(WIN32)
    add_executable(svr2011 WIN32 ${SVR2011_SOURCES} src/svr2011.rc)
elseif(ANDROID)
    # Android: libmain.so, which SDL's Java activity loads and starts (SDL_main).
    add_library(svr2011 SHARED ${SVR2011_SOURCES})
    set_target_properties(svr2011 PROPERTIES OUTPUT_NAME main)
    # Calls between the library's own functions (the recompiled game calls
    # its functions millions of times a second) bound directly, not through
    # the PLT (~3% of a phone's render thread in heavy entrances).
    target_compile_options(svr2011 PRIVATE -fno-semantic-interposition)
    target_link_options(svr2011 PRIVATE -Wl,-Bsymbolic-functions)
else()
    add_executable(svr2011 ${SVR2011_SOURCES})
endif()

rexglue_setup_target(svr2011 GPU_PLUGINS xenos)
if(SVR2011_P2P_SESSIONS)
    if(NOT WIN32)
        message(FATAL_ERROR "Enabled game sessions currently require the Windows HTTP transport")
    endif()
    if(NOT REXSDK_DIR OR NOT EXISTS "${REXSDK_DIR}/include/rex/system/xam/app_manager.h")
        message(FATAL_ERROR "Sessions require the source SDK with the repository's runtime patch applied")
    endif()
    file(READ "${REXSDK_DIR}/include/rex/system/xam/app_manager.h" session_sdk_header)
    if(NOT session_sdk_header MATCHES "SetSessionProvider")
        message(FATAL_ERROR "Reapply the updated port/patches/rexglue-sdk-svr2011.patch to a clean pinned SDK")
    endif()
    target_sources(svr2011 PRIVATE src/online_runtime.cpp src/online_http.cpp)
    target_compile_definitions(svr2011 PRIVATE SVR2011_P2P_SESSIONS=1)
    target_link_libraries(svr2011 PRIVATE svr2011_sessions winhttp)
endif()
# The SDK stages the GPU plugin (loaded at run time, never linked) only when
# the exe relinks: a change to the plugin alone would leave a stale copy
# beside the exe (and in Game Files / the release). Stage it on every build.
if(TARGET rexgpu-xenos AND NOT ANDROID)
    add_custom_target(svr2011_stage_gpu_plugin ALL
        COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:rexgpu-xenos> $<TARGET_FILE_DIR:svr2011>
        DEPENDS rexgpu-xenos svr2011
        VERBATIM)
endif()
if(NOT WIN32)
    # MSVC CRT functions the port uses (_dupenv_s ...).
    target_compile_options(svr2011 PRIVATE -include ${CMAKE_CURRENT_SOURCE_DIR}/src/compat.h)
endif()
if(SVR2011_D3D_TRACE)
    target_compile_definitions(svr2011 PRIVATE SVR2011_D3D_TRACE=1)
else()
    target_link_libraries(svr2011 PRIVATE plume)
endif()
if(REXSDK_DIR AND TARGET imgui)
    # SDK source tree: ImGui (built into the runtime) is not on the public
    # include path the way the installed SDK bundles it.
    target_include_directories(svr2011 PRIVATE
        $<TARGET_PROPERTY:imgui,INTERFACE_INCLUDE_DIRECTORIES>
        ${REXSDK_DIR}/thirdparty/xxHash)
endif()
if(WIN32)
    target_link_libraries(svr2011 PRIVATE xaudio2 ole32 winmm d3d12 dxgi dxguid dbghelp
        mfplat mfreadwrite mfuuid)
    # A PDB (function names + line tables) for crash reports and profiling.
    target_compile_options(svr2011 PRIVATE -gline-tables-only)
    target_link_options(svr2011 PRIVATE -Wl,/DEBUG)
    # The same for the SDK's runtime and GPU plugin (source-tree builds).
    foreach(sdk_target rexruntime rexgpu-xenos)
        if(TARGET ${sdk_target})
            target_compile_options(${sdk_target} PRIVATE -gline-tables-only)
            target_link_options(${sdk_target} PRIVATE -Wl,/DEBUG)
        endif()
    endforeach()
endif()

# The program icon (the game's title image) is made from your own game files
# by tools/make_icon.py; without it the programs are built without an icon.
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/assets/svr2011.ico")
    set_source_files_properties(src/svr2011.rc launcher/launcher.rc PROPERTIES COMPILE_DEFINITIONS SVR2011_ICON)
endif()

# -- Launcher -----------------------------------------------------------------
if(WIN32 AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/launcher/svr2011_launcher.c")
    add_executable(svr2011_launcher WIN32 launcher/svr2011_launcher.c launcher/apk_package.c launcher/movie_maker.c launcher/bink_decode.c launcher/updater.c launcher/unzip.c launcher/launcher.rc)
    set_target_properties(svr2011_launcher PROPERTIES OUTPUT_NAME "SvR2011 Launcher")
    target_compile_definitions(svr2011_launcher PRIVATE
        UNICODE _UNICODE _CRT_SECURE_NO_WARNINGS WIN32_LEAN_AND_MEAN)
    target_link_options(svr2011_launcher PRIVATE -Wl,/ENTRY:wWinMainCRTStartup)
    target_link_libraries(svr2011_launcher PRIVATE
        comctl32 comdlg32 shell32 ole32 uxtheme windowscodecs mfplat mfreadwrite mfuuid propsys winhttp)
    # The release version (VERSION.txt; package.ps1 names the zip after it and the
    # launcher's updater compares it with the newest GitHub release).
    file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION.txt" SVR2011_VERSION LIMIT_COUNT 1)
    target_compile_definitions(svr2011_launcher PRIVATE "PORT_VERSION=L\"${SVR2011_VERSION}\"")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION.txt")
endif()
